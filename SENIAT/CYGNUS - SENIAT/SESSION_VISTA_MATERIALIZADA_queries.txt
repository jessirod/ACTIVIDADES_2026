CREATE MATERIALIZED VIEW DBO.MV_DECLARACION_BASE
   TABLESPACE TS_D_MEDIUM
   NOLOGGING NOCACHE NOCOMPRESS
   BUILD IMMEDIATE
   REFRESH FAST ON DEMAND
AS
   SELECT d.ID_CONTRIBUYENTE      CONTRIBUYENTE,
          d.IMPUESTO_DECLARACION  IMPUESTO,
          d.PERIODO_DECLARACION   PERIODO,
          COUNT(*)                CANTIDAD          
     FROM DBO.DECLARACION d
    WHERE d.IMPUESTO_DECLARACION IN ('25','26','30')
      AND d.SITUACION_DECLARACION NOT IN ('00','31')
   GROUP BY d.ID_CONTRIBUYENTE,
            d.IMPUESTO_DECLARACION,
            d.PERIODO_DECLARACION;
