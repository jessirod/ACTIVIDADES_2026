-- EXPORTAR

DIRECTORY=IMPORT 
DUMPFILE=data_ordenes_especificas_%U.dmp 
LOGFILE=exp_ordenes_especificas.log 
SCHEMAS=PCP 
INCLUDE=TABLE:"IN ('ORDEN', 'ORDEN_DETAIL', 'DEBITS', 'TRACK_ORDEN', 'CREDITS_CREDITO_INMEDIATO', 'TORDEN', 'TORDEN_DETAIL', 'TORDEN_ALTA', 'TORDEN_DETAIL_ALTA', 'TRACK_LOTE')" 
QUERY=PCP.ORDEN:"WHERE id_orden >= 17087125" 
QUERY=PCP.ORDEN_DETAIL:"WHERE id_orden >= 17087125" 
QUERY=PCP.DEBITS:"WHERE id_orden >= 17087125" 
QUERY=PCP.TRACK_ORDEN:"WHERE TO_NUMBER(id_orden) >= 17087125" 
QUERY=PCP.CREDITS_CREDITO_INMEDIATO:"WHERE id_credits IN (SELECT c.id_credits FROM PCP.CREDITS c JOIN PCP.DEBITS d ON c.id_debits = d.id_debits WHERE d.id_orden >= 17087125)" 
QUERY=PCP.TORDEN:"WHERE id_torden >= 9928" 
QUERY=PCP.TORDEN_DETAIL:"WHERE id_torden >= 9928" 
QUERY=PCP.TORDEN_ALTA:"WHERE id_torden >= 1984" 
QUERY=PCP.TORDEN_DETAIL_ALTA:"WHERE id_torden >= 1984" 
QUERY=PCP.TRACK_LOTE:"WHERE id_lote IN (SELECT id_torden FROM PCP.TORDEN WHERE id_torden >= 9928 UNION SELECT id_torden FROM PCP.TORDEN_ALTA WHERE id_torden >= 1984)" 
PARALLEL=4
